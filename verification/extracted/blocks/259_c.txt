/* ITM printf redirection: retarget _write() or fputc() */
int _write(int file, char *ptr, int len)
{
    for (int i = 0; i < len; i++) {
        /* Wait until stimulus port 0 is ready */
        while (ITM->PORT[0].u32 == 0)
            ;
        ITM->PORT[0].u8 = (uint8_t)ptr[i];
    }
    return len;
}
