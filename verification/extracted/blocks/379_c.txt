typedef void (*CmdHandler_t)(const uint8_t *pucPayload,
                              uint16_t usLength);

typedef struct {
    uint8_t       ucCmdId;
    uint8_t       ucExpectedLen;   /* 0 = variable length */
    CmdHandler_t  pfHandler;
    const char   *pcName;          /* For debug logging */
} CmdDescriptor_t;
