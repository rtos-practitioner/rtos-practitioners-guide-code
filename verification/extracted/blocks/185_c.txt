/* uart_hal.c -- Vendor HAL wrapper */
#include "uart_hal.h"
#include "stm32f4xx_hal.h"

UART_HandleTypeDef huart2;

int uart_hal_init(uint32_t baudrate)
{
    /* Enable peripheral clocks */
    __HAL_RCC_USART2_CLK_ENABLE();
    __HAL_RCC_GPIOA_CLK_ENABLE();

    /* Configure GPIO: PA2 = TX (AF7), PA3 = RX (AF7) */
    GPIO_InitTypeDef gpio = {0};
    gpio.Pin       = GPIO_PIN_2 | GPIO_PIN_3;
    gpio.Mode      = GPIO_MODE_AF_PP;
    gpio.Pull      = GPIO_PULLUP;
    gpio.Speed     = GPIO_SPEED_FREQ_HIGH;
    gpio.Alternate = GPIO_AF7_USART2;
    HAL_GPIO_Init(GPIOA, &gpio);

    /* Configure UART */
    huart2.Instance          = USART2;
    huart2.Init.BaudRate     = baudrate;
    huart2.Init.WordLength   = UART_WORDLENGTH_8B;
    huart2.Init.StopBits     = UART_STOPBITS_1;
    huart2.Init.Parity       = UART_PARITY_NONE;
    huart2.Init.Mode         = UART_MODE_TX_RX;
    huart2.Init.HwFlowCtl    = UART_HWCONTROL_NONE;
    huart2.Init.OverSampling = UART_OVERSAMPLING_16;

    if (HAL_UART_Init(&huart2) != HAL_OK) {
        return -1;
    }

    /* Enable NVIC interrupt */
    HAL_NVIC_SetPriority(USART2_IRQn, 6, 0);  /* Below configMAX_SYSCALL */
    HAL_NVIC_EnableIRQ(USART2_IRQn);

    return 0;
}

/* Data-register access used by the RTOS layer (replaced by the fake HAL) */
uint8_t uart_hal_read_byte(void)
{
    return (uint8_t)(huart2.Instance->DR & 0xFF);
}

void uart_hal_write_byte(uint8_t byte)
{
    huart2.Instance->DR = byte;
}
